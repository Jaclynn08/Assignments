set.seed(42)
a<-function(n=numeric()){
x<- rnorm(n,0,2)
# the 3 estimators 
mu<-mean(x)
mu_min<- x-mu
sq_mu_min<-(mu_min)^2
sum_sq_mu_min<-sum(sq_mu_min)
#pop. var
sigma_hat_sq<- sum_sq_mu_min/(n)
#sample var
s_square<- sum_sq_mu_min/(n-1)
#standard dev. of sample
s<-sqrt(s_square)

return(c(sigma_hat_sq,s_square,s))
}
